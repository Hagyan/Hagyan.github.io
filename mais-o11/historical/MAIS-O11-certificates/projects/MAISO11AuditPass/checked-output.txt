def MAISO11.PolynomialLobBound : {F : Type} → MAISO11.ProofSystem F → (F → F → F) → (F → F) → (F → Nat) → Prop :=
fun {F} S imp box size =>
  ∃ C d, ∀ (A : F) (r : S.Proof (imp (box A) A)), ∃ p, S.length p ≤ C * (S.length r + size A + 1) ^ d

  {size : F → Nat} (T : MAISO11.QuantitativeLobTools S imp box size) (A : F) (r : S.Proof (imp (box A) A)) :
  S.length (T.convert A r) ≤
    S.length r + S.length (T.lob A) + S.length (T.notice A r) +
        T.mpCost * (size (box (imp (box A) A)) + size (box A) + 1) +
      T.mpCost * (size (box A) + size A + 1)
MAISO11.QuantitativeLobTools.budget {F : Type} {S : MAISO11.ProofSystem F} {imp : F → F → F} {box : F → F}
  {size : F → Nat} (T : MAISO11.QuantitativeLobTools S imp box size) (A : F) (r : S.Proof (imp (box A) A)) :
  S.length (T.convert A r) ≤
    S.length r + T.noticeCoeff * (S.length r + size A + 1) ^ T.noticeDegree + T.lobCoeff * (size A + 1) ^ T.lobDegree +
      T.assemblyCoeff * (size A + 1)
